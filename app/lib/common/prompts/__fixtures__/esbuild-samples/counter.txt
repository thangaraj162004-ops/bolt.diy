export function CounterButton({ label, onClick }: CounterButtonProps) {
  return (
    <button className="rounded-lg bg-indigo-600 px-4 py-2 font-semibold text-white hover:bg-indigo-500" onClick={onClick}>

<boltAction type="file" filePath="src/App.tsx">import { useState } from 'react';
import { CounterButton } from './components/CounterButton';

export default function App() {
  const [count, setCount] = useState(0);

  return (
    <main className="flex min-h-screen flex-col items-center justify-center gap-6 bg-slate-50">
      <h1 className="text-3xl font-bold text-slate-800">Counter</h1>
      <p className="text-6xl font-light tabular-nums">{count}</p>
      <div className="flex gap-3">
        <CounterButton label="−1" onClick={() => setCount((value) => value - 1)} />
        <CounterButton label="Reset" onClick={() => setCount(0)} />
        <CounterButton label="+1" onClick={() => setCount((value) => value + 1)} />
      </div>
    </main>
  );
}</boltAction>
