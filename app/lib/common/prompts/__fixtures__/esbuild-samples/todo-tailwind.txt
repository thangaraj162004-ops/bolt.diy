<boltAction type="file" filePath="src/hooks/useTodos.ts">import { useEffect, useState } from 'react';
import type { Todo } from '../types';

const STORAGE_KEY = 'todos';

function load(): Todo[] {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY) ?? '[]') as Todo[];
  } catch {
    return [];
  }
}

export function useTodos() {
  const [todos, setTodos] = useState<Todo[]>(load);

  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
    } catch {
      // storage can be unavailable in a sandboxed preview
    }
  }, [todos]);

  return {
    todos,
    add: (title: string) => setTodos((list) => [...list, { id: crypto.randomUUID(), title, done: false }]),
    toggle: (id: string) => setTodos((list) => list.map((t) => (t.id === id ? { ...t, done: !t.done } : t))),
    remove: (id: string) => setTodos((list) => list.filter((t) => t.id !== id)),
  };
}</boltAction>

import { Trash2 } from 'lucide-react';
import type { Todo } from '../types';

interface TodoItemProps {
  todo: Todo;
  onToggle: () => void;
  onRemove: () => void;
}

export function TodoItem({ todo, onToggle, onRemove }: TodoItemProps) {
  return (
    <li className="flex items-center gap-3 rounded-lg bg-white px-4 py-3 shadow-sm">
      <input type="checkbox" checked={todo.done} onChange={onToggle} className="h-4 w-4 accent-emerald-600" />
      <span className={clsx('flex-1', todo.done && 'text-slate-400 line-through')}>{todo.title}</span>
      <button aria-label={`Delete ${todo.title}`} onClick={onRemove} className="text-slate-400 hover:text-red-500">
        <Trash2 size={18} />
      </button>
    </li>
  );
}</boltAction>
<boltAction type="file" filePath="src/App.tsx">import { useState, type FormEvent } from 'react';
import { Plus } from 'lucide-react';
import { TodoItem } from './components/TodoItem';
import { useTodos } from './hooks/useTodos';

export default function App() {
  const { todos, add, toggle, remove } = useTodos();
  const [title, setTitle] = useState('');

  const submit = (event: FormEvent) => {
    event.preventDefault();

    if (title.trim()) {
      add(title.trim());
      setTitle('');
    }
  };

  return (
    <main className="mx-auto max-w-md p-6">
      <h1 className="mb-4 text-2xl font-bold">Todos</h1>
      <form onSubmit={submit} className="mb-4 flex gap-2">
        <input
          value={title}
          onChange={(event) => setTitle(event.target.value)}
          placeholder="What needs doing?"
          className="flex-1 rounded-lg border border-slate-300 px-3 py-2"
        />
        <button className="flex items-center gap-1 rounded-lg bg-emerald-600 px-3 py-2 text-white">
          <Plus size={18} /> Add
        </button>
      </form>
      <ul className="space-y-2">
        {todos.map((todo) => (
          <TodoItem key={todo.id} todo={todo} onToggle={() => toggle(todo.id)} onRemove={() => remove(todo.id)} />
        ))}
      </ul>
      <p className="mt-4 text-sm text-slate-500">{todos.filter((todo) => !todo.done).length} left</p>
    </main>
  );
}</boltAction>
